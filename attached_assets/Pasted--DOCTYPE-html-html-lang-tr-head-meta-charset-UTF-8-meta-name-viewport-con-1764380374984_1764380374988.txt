        function userLogin() {
            const companyId = document.getElementById('user-company').value;
            const username = document.getElementById('user-username').value;
            const password = document.getElementById('user-password').value;
            
            if (!companyId || !username || !password) {
                showError('Lütfen tüm alanları doldurun.');
                return;
            }
            
            const company = companies.find(c => c.id === companyId);
            if (!company) {
                showError('Şirket bulunamadı.');
                return;
            }
            
            // Admin girişi kontrolü
            if (company.admin.username === username && company.admin.password === password) {
                currentUser = { username, isAdmin: true };
                currentCompany = company;
                showMainApp();
                return;
            }
            
            // Normal kullanıcı girişi kontrolü
            const user = company.users.find(u => u.username === username && u.password === password);
            if (user) {
                currentUser = { ...user, isAdmin: false };
                currentCompany = company;
                showMainApp();
                return;
            }
            
            showError('Kullanıcı adı veya şifre hatalı.');
        }

        function downloadDashboardReport() {
            const userActivities = currentCompany.activities ? currentCompany.activities.filter(a => a.user === currentUser.username) : [];
            
            if (userActivities.length === 0) {
                showError('Henüz aktivite bulunmamaktadır.');
                return;
            }
            
            let report = `Dashboard Raporu - ${new Date().toLocaleDateString('tr-TR')}\n\n`;
            report += `Kullanıcı: ${currentUser.username}\n`;
            report += `Şirket: ${currentCompany.name}\n\n`;
            
            report += `Toplam Mesaj: ${currentCompany.messages ? currentCompany.messages.length : 0}\n`;
            report += `Performans Puanı: ${userActivities.reduce((sum, activity) => sum + (activity.points || 0), 0)}\n`;
            report += `Aktif Kullanıcılar: ${currentCompany.users.length + 1}\n`;
            report += `Tamamlanan Görevler: ${userActivities.length}\n\n`;
            
            report += `Son Aktiviteler:\n`;
            userActivities.slice(-5).reverse().forEach(activity => {
                report += `- ${getActivityName(activity.type)}: ${activity.points} puan (${formatTime(activity.startTime)})\n`;
            });
            
            // Raporu indir
            const blob = new Blob([report], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `dashboard-raporu-${new Date().toISOString().split('T')[0]}.txt`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            showSuccess('Dashboard raporu başarıyla indirildi.');
        }

        function handleFileSelect(e) {
            const files = e.target.files;
            
            if (files.length === 0) return;
            
            selectedFiles = Array.from(files);
            
            // Dosya bilgilerini göster
            const fileUploadArea = document.getElementById('file-upload-area');

        function sendFiles() {
            if (selectedFiles.length === 0) return;
            
            selectedFiles.forEach(file => {
                const reader = new FileReader();
                
                reader.onload = function(e) {
                    const fileData = {
                        id: Date.now().toString() + Math.random(),
                        name: file.name,
                        type: file.type.split('/')[0],
                        size: formatFileSize(file.size),
                        data: e.target.result,
                        sender: currentUser.username,
                        timestamp: new Date().toISOString()
                    };
                    
                    if (!currentCompany.files) currentCompany.files = [];
                    currentCompany.files.push(fileData);
                    
                    // Mesaj olarak dosya paylaşımını kaydet
                    const message = {
                        id: Date.now().toString(),
                        sender: currentUser.username,
                        fileName: file.name,
                        fileId: fileData.id,
                        timestamp: new Date().toISOString(),
                        type: 'file'
                    };
                    
                    if (!currentCompany.messages) currentCompany.messages = [];
                    currentCompany.messages.push(message);
                    
                    localStorage.setItem('companies', JSON.stringify(companies));
                    
                    // Dosya listesini güncelle
                    updateFileList();
                    
                    // Sohbet mesajlarını güncelle
                    loadChatMessages();
                };
                
                reader.readAsDataURL(file);
            });
            
            // Dosya yükleme alanını gizle
            document.getElementById('file-upload-area').style.display = 'none';
            selectedFiles = [];
            
            showSuccess('Dosya(lar) başarıyla paylaşıldı.');
        }

        function formatFileSize(bytes) {
            if (bytes === 0) return '0 Bytes';
            const k = 1024;
            const sizes = ['Bytes', 'KB', 'MB', 'GB'];
            const i = Math.floor(Math.log(bytes) / Math.log(k));
            return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
        }
        
        function startShift() {
            if (!navigator.geolocation) {
                showError('Konum bilgisi alınamıyor. Tarayıcınız bu özelliği desteklemiyor.');
                return;
            }
            
            navigator.geolocation.getCurrentPosition(
                function(position) {
                    const now = new Date();
                    currentShift = {
                        userId: currentUser.username,
                        startTime: now,
                        startLocation: {
                            lat: position.coords.latitude,
                            lng: position.coords.longitude
                        },
                        activities: []
                    };
                    
                    // Mesai bilgilerini güncelle
                    document.getElementById('shift-info').innerHTML = `
                        <p><strong>Mesai Başlangıcı:</strong> ${now.toLocaleString('tr-TR')}</p>
                        <p><strong>Konum:</strong> ${position.coords.latitude.toFixed(6)}, ${position.coords.longitude.toFixed(6)}</p>
                    `;
                    
                    // Süre gösterimini aktif et
                    document.getElementById('shift-duration').style.display = 'block';
                    
                    // Butonları güncelle
                    document.getElementById('start-shift-btn').disabled = true;
                    document.getElementById('end-shift-btn').disabled = false;
                    
                    // Süre sayacını başlat
                    shiftInterval = setInterval(updateShiftTimer, 1000);
                    
                    showSuccess('Mesai başarıyla başlatıldı.');
                },
                function(error) {
                    showError('Konum bilgisi alınamadı: ' + error.message);
                }
            );
        }
        
        function updateShiftTimer() {
            if (!currentShift) return;
            
            const now = new Date();
            const diff = now - currentShift.startTime;
            const hours = Math.floor(diff / 3600000);
            const minutes = Math.floor((diff % 3600000) / 60000);
            const seconds = Math.floor((diff % 60000) / 1000);
            
            document.getElementById('shift-timer').textContent = 
                `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
        }

        function startActivity() {
            const activityType = document.getElementById('activity-type').value;

                notes: document.getElementById('activity-notes').value
            };
            
            // Aktivite zamanlayıcısını başlat
            document.getElementById('activity-timer').style.display = 'block';
            activityInterval = setInterval(updateActivityTimer, 1000);

        function updateActivityTimer() {
            if (!currentActivity) return;
            
            const now = new Date();
            const diff = now - currentActivity.startTime;
            const hours = Math.floor(diff / 3600000);
            const minutes = Math.floor((diff % 3600000) / 60000);
            const seconds = Math.floor((diff % 60000) / 1000);
            
            document.getElementById('activity-duration').textContent = 
                `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;

        function stopActivity() {
            if (!currentActivity) return;
            
            const now = new Date();
            currentActivity.endTime = now;
            currentActivity.duration = (now - currentActivity.startTime) / (1000 * 60 * 60); // saat cinsinden
            
            // Aktivite zamanlayıcısını durdur
            clearInterval(activityInterval);
            
            // Butonları güncelle
            document.getElementById('stop-activity-btn').disabled = true;
            document.getElementById('save-activity-btn').disabled = false;
            
            showSuccess('Aktivite durduruldu. Kaydet butonuna basarak kaydedebilirsiniz.');
        }
        
        function saveActivity() {
            if (!currentActivity) return;
            
            // Puanı hesapla
            currentActivity.points = calculateActivityPoints(currentActivity.type, currentActivity.duration);
            currentActivity.date = new Date().toISOString().split('T')[0];
            
            // Aktiviteleri kaydet
            if (!currentCompany.activities) currentCompany.activities = [];
            currentCompany.activities.push(currentActivity);
            localStorage.setItem('companies', JSON.stringify(companies));
            
            // Formu temizle
            document.getElementById('customer-name').value = '';
            document.getElementById('activity-notes').value = '';
            document.getElementById('activity-timer').style.display = 'none';
            
            // Butonları sıfırla
            document.getElementById('start-activity-btn').disabled = false;
            document.getElementById('save-activity-btn').disabled = true;
            
            // Aktiviteleri güncelle
            updateActivityList();
            
            showSuccess('Aktivite başarıyla kaydedildi. Puan: ' + currentActivity.points);
            
            currentActivity = null;
        }
        
        function calculateActivityPoints(activityType, duration) {
            const basePoints = currentCompany.scoring[activityType] || 1;
            return Math.round(basePoints * duration * 10) / 10;
        }

        function generateDailyReport() {
            const userActivities = currentCompany.activities ? currentCompany.activities.filter(a => a.user === currentUser.username && a.date === new Date().toISOString().split('T')[0]) : [];
            
            if (userActivities.length === 0) {
                showError('Bugün için kayıtlı aktivite bulunmamaktadır.');
                return;
            }
            
            let report = `Günlük Performans Raporu - ${new Date().toLocaleDateString('tr-TR')}\n\n`;
            report += `Kullanıcı: ${currentUser.username}\n`;
            report += `Departman: ${getDepartmentName(selectedDepartment)}\n\n`;
            
            let totalPoints = 0;
            userActivities.forEach(activity => {
                report += `• ${getActivityName(activity.type)}: ${activity.points} puan (${activity.duration.toFixed(1)} saat)\n`;
                if (activity.customerName) {
                    report += `  Müşteri: ${activity.customerName}\n`;
                }
                if (activity.notes) {
                    report += `  Not: ${activity.notes}\n`;
                }
                report += '\n';
                totalPoints += activity.points;
            });
            
            report += `TOPLAM PUAN: ${totalPoints.toFixed(1)}\n`;
            
            // Raporu indir
            const blob = new Blob([report], { type: 'text/plain' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `performans-raporu-${new Date().toISOString().split('T')[0]}.txt`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            showSuccess('Günlük rapor başarıyla oluşturuldu ve indirildi.');
        }

            setupAdminControls();
            showSuccess('Kullanıcı başarıyla silindi.');
        }
        
        function logout() {
            if (confirm('Çıkış yapmak istediğinizden emin misiniz?')) {
                // Süre sayaçlarını durdur
                if (shiftInterval) clearInterval(shiftInterval);
                if (activityInterval) clearInterval(activityInterval);
                
                // Ana uygulamayı gizle
                document.getElementById('main-app').style.display = 'none';
                
                // Giriş ekranını göster
                document.getElementById('login-screen').style.display = 'flex';
                
                // Formları temizle
                document.getElementById('company-name').value = '';
                document.getElementById('admin-username').value = '';
                document.getElementById('admin-password').value = '';
                document.getElementById('user-username').value = '';
                document.getElementById('user-password').value = '';
                
                // Mesajları gizle
                document.getElementById('success-message').style.display = 'none';
                document.getElementById('error-message').style.display = 'none';
                
                // Giriş sekmesini sıfırla
                document.querySelectorAll('.login-tab').forEach(t => t.classList.remove('active'));
                document.querySelector('[data-login-tab="company"]').classList.add('active');
                document.querySelectorAll('.login-content').forEach(c => c.classList.add('hidden'));
                document.getElementById('company-login').classList.remove('hidden');
                
                // Değişkenleri sıfırla
                currentUser = null;
                currentCompany = null;
                currentShift = null;
                currentActivity = null;
            }
        }
        
        function showSuccess(message) {
            const successElement = document.getElementById('success-message');
            successElement.textContent = message;
            successElement.style.display = 'block';
            
            // Hata mesajını gizle
            document.getElementById('error-message').style.display = 'none';
            
            // 5 saniye sonra gizle
            setTimeout(() => {
                successElement.style.display = 'none';
            }, 5000);
        }
        
        function showError(message) {
            const errorElement = document.getElementById('error-message');
            errorElement.textContent = message;
            errorElement.style.display = 'block';
            
            // Başarı mesajını gizle
            document.getElementById('success-message').style.display = 'none';
            
            // 5 saniye sonra gizle
            setTimeout(() => {
                errorElement.style.display = 'none';
            }, 5000);
        }
